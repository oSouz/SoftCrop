cmake_minimum_required(VERSION 3.20)
project(SouzSoftCrop VERSION 1.0 LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Build requires Windows x64")
endif()
set(FFGL_SDK "${CMAKE_CURRENT_SOURCE_DIR}/vendor/ffgl" CACHE PATH "Official FFGL v2.1 SDK")
if(NOT EXISTS "${FFGL_SDK}/source/lib/FFGLSDK.h")
  message(FATAL_ERROR "Obtain official resolume/ffgl tag v2.1 in vendor/ffgl first")
endif()
file(GLOB FFGL_CORE CONFIGURE_DEPENDS "${FFGL_SDK}/source/lib/ffgl/*.cpp")
file(GLOB FFGL_UTIL CONFIGURE_DEPENDS "${FFGL_SDK}/source/lib/ffglex/*.cpp")
add_library(SouzSoftCrop_FFGL21 SHARED src/SouzSoftCrop.cpp ${FFGL_CORE} ${FFGL_UTIL}
  "${FFGL_SDK}/source/lib/glsdk_0_5_2/glload/source/gl_load.c"
  "${FFGL_SDK}/build/windows/FFGLPlugins.def")
target_include_directories(SouzSoftCrop_FFGL21 PRIVATE "${FFGL_SDK}/source/lib")
target_compile_features(SouzSoftCrop_FFGL21 PRIVATE cxx_std_17)
target_compile_definitions(SouzSoftCrop_FFGL21 PRIVATE _CRT_SECURE_NO_WARNINGS)
target_link_libraries(SouzSoftCrop_FFGL21 PRIVATE opengl32)
set_target_properties(SouzSoftCrop_FFGL21 PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
